cmake_minimum_required(VERSION 3.25)
project(MavlinkGateway VERSION 1.0.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT MINGW)
    message(FATAL_ERROR "This version targets Windows with MinGW-w64 GCC (MSYS2 UCRT64). Linux support is not implemented yet.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_SOURCE_DIR}/build/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/lib")
option(MGW_CLIENT_LOGGING "Enable CSV logging in the C client" ON)

add_library(mgw_options INTERFACE)
target_compile_options(mgw_options INTERFACE -Wall -Wextra -Wpedantic -pthread)
target_link_options(mgw_options INTERFACE -static -pthread)
target_compile_definitions(mgw_options INTERFACE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(mgw_options INTERFACE "${PROJECT_SOURCE_DIR}/include")

add_library(mavlink_headers INTERFACE)
target_include_directories(mavlink_headers SYSTEM INTERFACE "${PROJECT_SOURCE_DIR}/third_party/mavlink")
add_library(nlohmann_headers INTERFACE)
target_include_directories(nlohmann_headers SYSTEM INTERFACE "${PROJECT_SOURCE_DIR}/third_party")

# Record the actual source content, independent of the checkout's absolute path.
file(GLOB_RECURSE fingerprint_inputs CONFIGURE_DEPENDS
    "${PROJECT_SOURCE_DIR}/apps/*.cpp" "${PROJECT_SOURCE_DIR}/src/*.cpp"
    "${PROJECT_SOURCE_DIR}/src/*.hpp" "${PROJECT_SOURCE_DIR}/include/*.hpp"
    "${PROJECT_SOURCE_DIR}/third_party/*.h" "${PROJECT_SOURCE_DIR}/third_party/*.hpp")
list(APPEND fingerprint_inputs "${PROJECT_SOURCE_DIR}/CMakeLists.txt" "${PROJECT_SOURCE_DIR}/CMakePresets.json")
list(SORT fingerprint_inputs)
set(source_manifest "")
foreach(input IN LISTS fingerprint_inputs)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${input}")
    file(RELATIVE_PATH relative_name "${PROJECT_SOURCE_DIR}" "${input}")
    file(SHA256 "${input}" input_hash)
    string(APPEND source_manifest "${relative_name}:${input_hash}\n")
endforeach()
string(SHA256 source_hash "${source_manifest}")
file(MAKE_DIRECTORY "${PROJECT_BINARY_DIR}/generated")
file(CONFIGURE OUTPUT "${PROJECT_BINARY_DIR}/generated/client_build.hpp"
    CONTENT "#pragma once\n#define C_CLIENT_SOURCE_SHA256 \"@source_hash@\"\n" @ONLY)

add_library(mgw_processing STATIC src/processing/processing.cpp)
target_link_libraries(mgw_processing PUBLIC mgw_options PRIVATE mavlink_headers)
add_library(mgw_logging STATIC src/logging/csv_logger.cpp)
target_link_libraries(mgw_logging PUBLIC mgw_processing)
add_library(mgw_config STATIC src/config/config.cpp)
target_link_libraries(mgw_config PUBLIC mgw_options PRIVATE nlohmann_headers)

add_library(mgw_gateway STATIC src/gateway/network.cpp src/gateway/audit.cpp src/gateway/gateway.cpp)
target_include_directories(mgw_gateway PUBLIC src/gateway)
target_link_libraries(mgw_gateway PUBLIC mgw_config mgw_logging PRIVATE nlohmann_headers ws2_32 psapi)

add_library(mgw_transport STATIC src/client/net_socket.cpp src/client/send_queue.cpp)
target_include_directories(mgw_transport PUBLIC src/client)
target_link_libraries(mgw_transport PUBLIC mgw_options PRIVATE ws2_32)
add_library(mgw_client STATIC
    src/client/client_config.cpp src/client/client_metrics.cpp src/client/client_log.cpp
    src/client/mavlink_builder.cpp src/client/command_manager.cpp)
target_include_directories(mgw_client PUBLIC src/client PRIVATE "${PROJECT_BINARY_DIR}/generated")
target_link_libraries(mgw_client PUBLIC mgw_config mgw_processing mgw_transport PRIVATE nlohmann_headers mavlink_headers psapi)
if(MGW_CLIENT_LOGGING)
    target_link_libraries(mgw_client PRIVATE mgw_logging)
else()
    target_compile_definitions(mgw_client PUBLIC C_NO_LOGGER)
endif()

add_executable(gateway apps/gateway_main.cpp)
target_link_libraries(gateway PRIVATE mgw_gateway)
add_executable(c_client apps/client_main.cpp)
target_link_libraries(c_client PRIVATE mgw_client)
add_executable(fake_gateway tools/fake_gateway.cpp)
target_link_libraries(fake_gateway PRIVATE mgw_transport mgw_processing mavlink_headers)
add_executable(replay_udp tools/replay_udp.cpp)
target_link_libraries(replay_udp PRIVATE mgw_transport mgw_processing)
add_executable(fixture_generator tools/generate_fixtures.cpp)
target_link_libraries(fixture_generator PRIVATE mgw_options mavlink_headers)
add_executable(b_demo examples/parser_demo.cpp)
target_link_libraries(b_demo PRIVATE mgw_logging)

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
    add_executable(config_tests tests/unit/config/config_tests.cpp)
    target_link_libraries(config_tests PRIVATE mgw_config nlohmann_headers)
    add_executable(network_primitives_tests tests/unit/gateway/network_primitives_tests.cpp)
    target_link_libraries(network_primitives_tests PRIVATE mgw_gateway)
    add_executable(b_tests tests/unit/processing/processing_tests.cpp)
    target_link_libraries(b_tests PRIVATE mgw_logging)
    add_executable(a_tests tests/unit/gateway/gateway_tests.cpp)
    target_link_libraries(a_tests PRIVATE mgw_gateway)
    add_executable(c_tests tests/unit/client/c_tests.cpp)
    target_link_libraries(c_tests PRIVATE mgw_client)
    add_executable(metrics_tests tests/unit/client/metrics_tests.cpp)
    target_link_libraries(metrics_tests PRIVATE mgw_client)
    add_executable(send_queue_tests tests/unit/client/send_queue_tests.cpp)
    target_link_libraries(send_queue_tests PRIVATE mgw_transport)

    add_test(NAME fixture_verification COMMAND "${Python3_EXECUTABLE}" -B "${PROJECT_SOURCE_DIR}/tests/verify_fixtures.py" "${PROJECT_SOURCE_DIR}/tests/fixtures/v1")
    foreach(test IN ITEMS config_tests network_primitives_tests b_tests a_tests c_tests metrics_tests send_queue_tests)
        add_test(NAME ${test} COMMAND $<TARGET_FILE:${test}>)
        set_tests_properties(${test} PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" TIMEOUT 30 LABELS unit)
    endforeach()
    set_tests_properties(fixture_verification PROPERTIES LABELS unit TIMEOUT 10)

    function(add_network_test name script)
        add_test(NAME ${name} COMMAND "${Python3_EXECUTABLE}" -B "${PROJECT_SOURCE_DIR}/tests/run_integration.py" ${name} "${PROJECT_SOURCE_DIR}/${script}" ${ARGN})
        set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" TIMEOUT 300 RUN_SERIAL TRUE LABELS integration)
    endfunction()
    add_network_test(gateway_integration tests/integration/gateway_test.py)
    add_test(NAME abc_transport_integration COMMAND "${Python3_EXECUTABLE}" -B "${PROJECT_SOURCE_DIR}/tests/integration/abc_transport_test.py")
    set_tests_properties(abc_transport_integration PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" TIMEOUT 60 RUN_SERIAL TRUE LABELS integration)
    if(MGW_CLIENT_LOGGING)
        foreach(test IN ITEMS command_retry command_ack client_config fake_gateway client_metrics)
            add_network_test(${test}_integration tests/integration/${test}_test.py)
        endforeach()
        foreach(mode IN ITEMS B0 B1 B2)
            add_network_test(abc_${mode} tools/local_demo.py --mode ${mode} --seconds 4 --observers 2)
        endforeach()
    endif()
endif()
